03 THE FRAMEWORKS

One engine.
Your stack.

Use the core tour directly, or let a small provider expose it to your React or Vue tree.

REACTvirtual-site-tour-react
import { TourProvider, useTour } from "virtual-site-tour-react";
import type { TourConfig } from "virtual-site-tour";
import SettingsPage from "./SettingsPage";

const tour = {
  name: "team-settings",
  steps: [
    {
      id: "invite",
      title: "Invite your team",
      body: "Add a teammate here.",
      target: "#invite-member",
    },
  ],
} satisfies TourConfig;

function TourButton() {
  const { start, isRunning } = useTour();
  return (
    <button disabled={isRunning} onClick={() => void start()}>
      Show me around
    </button>
  );
}

export function App() {
  return (
    <TourProvider config={tour}>
      <SettingsPage />
      <TourButton />
    </TourProvider>
  );
}

The page target and hook component live inside the provider.

VUEvirtual-site-tour-vue
App.vue
<script setup lang="ts">
import { TourProvider } from "virtual-site-tour-vue";
import type { TourConfig } from "virtual-site-tour";
import SettingsPage from "./SettingsPage.vue";
import TourButton from "./TourButton.vue";

const tour: TourConfig = {
  name: "team-settings",
  steps: [
    {
      id: "invite",
      title: "Invite your team",
      body: "Add a teammate here.",
      target: "#invite-member",
    },
  ],
};
</script>

<template>
  <TourProvider :config="tour">
    <SettingsPage />
    <TourButton />
  </TourProvider>
</template>
TourButton.vue
<script setup lang="ts">
import { useTour } from "virtual-site-tour-vue";
const { start, isRunning, currentStep } = useTour();
</script>

<template>
  <button :disabled="isRunning" @click="start">
    {{ currentStep ? `Step: ${currentStep.id}` : "Show me around" }}
  </button>
</template>

Call useTour() inside a provider descendant.
